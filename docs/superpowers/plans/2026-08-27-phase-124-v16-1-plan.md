```python
from __future__ import annotations

import re
from pathlib import Path

REPO_ROOT = Path(__file__).resolve().parents[1]


def _read_toml(path: Path) -> str:
    return path.read_text(encoding="utf-8")


def _extract_section(text: str, header: str) -> str | None:
    pattern = re.compile(rf"^\[{re.escape(header)}\]\s*(.*?)(?=^\[|\Z)", re.MULTILINE | re.DOTALL)
    m = pattern.search(text)
    return m.group(1).strip() if m else None


def test_lingwen_shared_in_workspace_members() -> None:
    """Root pyproject.toml must list packages/lingwen-shared as a workspace member."""
    text = _read_toml(REPO_ROOT / "pyproject.toml")
    section = _extract_section(text, "tool.uv.workspace")
    assert section is not None, "root pyproject.toml missing [tool.uv.workspace]"
    assert "packages/lingwen-shared" in section, (
        f"workspace missing packages/lingwen-shared — found section:\n{section}"
    )


def test_lingwen_shared_directory_exists() -> None:
    """packages/lingwen-shared/ directory must exist."""
    assert (REPO_ROOT / "packages" / "lingwen-shared").is_dir(), (
        "packages/lingwen-shared/ directory missing"
    )


def test_lingwen_shared_pyproject_exists() -> None:
    """packages/lingwen-shared/pyproject.toml must exist."""
    path = REPO_ROOT / "packages" / "lingwen-shared" / "pyproject.toml"
    assert path.is_file(), f"{path} missing"


def test_lingwen_shared_pyproject_declares_project_section() -> None:
```

- [ ] **T2.3: 写 world DTO** — `packages/lingwen-shared/src/lingwen_shared/contracts/python/world.py`

```python
"""Pydantic v2 DTOs for the world bounded context.

Source of truth: lingwen_shared.contracts.python.world (this file).
TypeScript types are auto-generated by tooling/contracts/generate.py.

This file is the v16.1 starter set. Additions go here, not in routes/world.py
(which now imports from this module and re-exports for backward compat).
"""
from __future__ import annotations

from typing import Literal, Mapping, Any
from pydantic import BaseModel, Field, ConfigDict


# ---------------------------------------------------------------------------
# Character
# ---------------------------------------------------------------------------


class CharacterDTO(BaseModel):
    """Character row from the world DB (characters table).

    Mirrors ``infra/world_db/queries/characters.py::row_to_dict`` output shape.
    """

    model_config = ConfigDict(extra="ignore")

    id: int
    slug: str
    name: str
    canon_level: Literal["Draft", "Secondary", "Primary"]
    status: str | None = None
    first_chapter: int | None = None
    last_seen_chapter: int | None = None
    attributes: Mapping[str, Any] | None = None
    aliases: list[str] | None = None
    notes: str | None = None


# ---------------------------------------------------------------------------
# Faction
# ---------------------------------------------------------------------------


class FactionDTO(BaseModel):
    """Faction row from the world DB (faction table)."""

    model_config = ConfigDict(extra="ignore")

    id: int
    slug: str
    name: str
    description: str | None = None
    canon_level: Literal["Draft", "Secondary", "Primary"] = "Draft"


# ---------------------------------------------------------------------------
# Lore
# ---------------------------------------------------------------------------


class LoreDTO(BaseModel):
    """Lore (world book entry) row from the world DB (lore table)."""

    model_config = ConfigDict(extra="ignore")

    id: int
    category: str
    title: str
    body: str
    canon_level: Literal["Draft", "Secondary", "Primary"] = "Draft"


# ---------------------------------------------------------------------------
# Timeline event
# ---------------------------------------------------------------------------


class TimelineEventDTO(BaseModel):
    """Timeline event row from the world DB (timeline_event table)."""

    model_config = ConfigDict(extra="ignore")

    id: int
    project_slug: str
    chapter_number: int
    title: str
    description: str | None = None
    characters_involved: list[int] | None = None
    canon_level: Literal["Draft", "Secondary", "Primary"] = "Draft"


# ---------------------------------------------------------------------------
# Proposal (LLM agent extraction result)
```

```python
"""Pydantic → TypeScript codegen for lingwen-shared contracts.

Usage:
    uv run python tooling/contracts/generate.py

Reads:
    packages/lingwen-shared/src/lingwen_shared/contracts/python/{world,workspace,quality}.py

Writes:
    packages/lingwen-shared/src/lingwen_shared/contracts/ts/{world,workspace,quality}.ts

Uses pydantic-to-typescript CLI invocation. Do not edit generated TS manually.
"""
from __future__ import annotations

import json
import shutil
import subprocess
import sys
from pathlib import Path

REPO_ROOT = Path(__file__).resolve().parents[2]
SHARED_SRC = REPO_ROOT / "packages" / "lingwen-shared" / "src" / "lingwen_shared"
PYTHON_DIR = SHARED_SRC / "contracts" / "python"
TS_DIR = SHARED_SRC / "contracts" / "ts"

MODULES = ["world", "workspace", "quality"]


def _emit_json_schema(module_name: str) -> Path:
    """Run pydantic-to-typescript's json-schema emission to a tmp file.

    Uses pydantic v2's model_json_schema() directly (avoids external CLI
    version drift). One JSON Schema per module → TS interface emission done
    by _ts_from_json_schema.
    """
    import importlib

    mod = importlib.import_module(f"lingwen_shared.contracts.python.{module_name}")
    schema: dict = {"$defs": {}, "$ref": f"#/$defs/Root"}
    for name in dir(mod):
        obj = getattr(mod, name)
        if isinstance(obj, type) and hasattr(obj, "model_json_schema"):
            try:
                sub = obj.model_json_schema()
                schema["$defs"][name] = sub
            except Exception:
                continue
    out = Path(f"/tmp/lingwen_shared_{module_name}_schema.json")
    out.write_text(json.dumps(schema, indent=2), encoding="utf-8")
    return out


def _ts_from_json_schema(module_name: str, schema_path: Path) -> str:
    """Convert a Pydantic JSON Schema dump into a hand-rolled TS module.

    Strategy: walk $defs, emit one `export interface X { ... }` per Pydantic
    model with property name → TS type. Nested models become references.
    This is intentionally minimal — full zod reverse validation (§T5) catches
    drift, so the TS need not be a perfect mirror.
    """
    schema = json.loads(schema_path.read_text(encoding="utf-8"))
    defs: dict = schema.get("$defs", {})

    def _ts_type_for(prop_schema: dict, defs_lookup: dict) -> str:
        t = prop_schema.get("type")
        if t == "string":
            enum = prop_schema.get("enum")
            if enum:
                return " | ".join(f'"{v}"' for v in enum)
            return "string"
        if t == "integer":
            return "number"
        if t == "number":
            return "number"
        if t == "boolean":
            return "boolean"
        if t == "array":
            item_t = _ts_type_for(prop_schema.get("items", {}), defs_lookup)
            return f"{item_t}[]"
        if t == "object":
            ref = prop_schema.get("$ref")
            if ref and ref.startswith("#/$defs/"):
                return ref.split("/")[-1]
            return "Record<string, unknown>"
        if "anyOf" in prop_schema:
            parts = [_ts_type_for(s, defs_lookup) for s in prop_schema["anyOf"]]
            parts = [p for p in parts if p != "null"]
            return " | ".join(parts) + " | null" if "null" in [p for p in parts] else " | ".join(parts)
        return "unknown"

    lines = [
        "// AUTO-GENERATED by tooling/contracts/generate.py — do not edit.",
        f"// Source: packages/lingwen-shared/src/lingwen_shared/contracts/python/{module_name}.py",
        "",
    ]
    for name, def_schema in defs.items():
        if not isinstance(def_schema, dict) or def_schema.get("type") != "object":
            continue
        lines.append(f"export interface {name} {{")
        props = def_schema.get("properties", {})
        required = set(def_schema.get("required", []))
        for prop_name, prop_schema in props.items():
            opt = "" if prop_name in required else "?"
            ts_type = _ts_type_for(prop_schema, defs)
            lines.append(f"  {prop_name}{opt}: {ts_type};")
        lines.append("}")
        lines.append("")
    return "\n".join(lines)


def generate() -> int:
    for module_name in MODULES:
        py_file = PYTHON_DIR / f"{module_name}.py"
        if not py_file.is_file():
            print(f"SKIP {module_name}: {py_file} not found")
            continue
        schema_path = _emit_json_schema(module_name)
        ts_content = _ts_from_json_schema(module_name, schema_path)
        out = TS_DIR / f"{module_name}.ts"
        out.parent.mkdir(parents=True, exist_ok=True)
        out.write_text(ts_content, encoding="utf-8")
        print(f"WROTE {out.relative_to(REPO_ROOT)} ({len(ts_content)} bytes)")
    return 0


def main() -> int:
    return generate()


if __name__ == "__main__":
    sys.exit(main())
```

```python
"""Dump FastAPI OpenAPI schema to a JSON file for zod reverse validation.

Usage:
    uv run python tooling/contracts/dump_openapi.py --base-url http://localhost:8765 --out /tmp/openapi.json

Or against a saved dump:
    uv run python tooling/contracts/dump_openapi.py --from-file /tmp/openapi.json --out /tmp/openapi.json
"""
from __future__ import annotations

import argparse
import json
import sys
from pathlib import Path
from urllib.request import urlopen


def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--base-url", default="http://localhost:8765")
    parser.add_argument("--out", default="/tmp/lingwen_openapi.json")
    parser.add_argument("--from-file", help="use a pre-saved openapi.json instead of fetching")
    args = parser.parse_args()

    if args.from_file:
        data = json.loads(Path(args.from_file).read_text(encoding="utf-8"))
    else:
        url = f"{args.base_url.rstrip('/')}/openapi.json"
        try:
            with urlopen(url, timeout=10) as resp:
                data = json.loads(resp.read().decode("utf-8"))
        except Exception as exc:
            print(f"FAIL: cannot fetch OpenAPI from {url}: {exc}", file=sys.stderr)
            return 1

    Path(args.out).write_text(json.dumps(data, indent=2), encoding="utf-8")
    print(f"WROTE {args.out}")
    return 0


if __name__ == "__main__":
    sys.exit(main())
```

```python
"""zod reverse validation for lingwen-shared contracts.

Compares Pydantic-derived TypeScript types against zod schemas generated
from the FastAPI OpenAPI schema. Detects drift between Python source of
truth and the live API contract.

v16.1 状态: 作为独立 CI job 运行 (Q5=B),不在 dev loop 跑。

Usage:
    # 1. dump OpenAPI from running backend
    uv run python tooling/contracts/dump_openapi.py --out /tmp/openapi.json
    # 2. compare
    uv run python tooling/contracts/zod_revalidate.py --openapi /tmp/openapi.json
"""
from __future__ import annotations

import argparse
import json
import re
import subprocess
import sys
from pathlib import Path

REPO_ROOT = Path(__file__).resolve().parents[2]
SHARED_TS_DIR = (
    REPO_ROOT
    / "packages"
    / "lingwen-shared"
    / "src"
    / "lingwen_shared"
    / "contracts"
    / "ts"
)


def _extract_required_fields_from_ts(ts_content: str, interface_name: str) -> set[str]:
    """Extract required field names from a TS interface.

    Heuristic: any property without `?` after the colon is required.
    Properties listed in `Required<Pick<X, ...>>` utility types are also
    detected but not present in our minimal codegen, so we ignore them.
    """
    pattern = re.compile(
        rf"interface\s+{interface_name}\s*\{{(.*?)\}}",
        re.DOTALL,
    )
    m = pattern.search(ts_content)
    if not m:
        return set()
    body = m.group(1)
    required: set[str] = set()
    for line in body.splitlines():
        line = line.strip()
        if not line or line.startswith("//") or line.startswith("/*"):
            continue
        # e.g. `id: number;` (required) or `name?: string;` (optional)
        m2 = re.match(r"(\w+)(\??)\s*:", line)
        if not m2:
            continue
        field, opt = m2.group(1), m2.group(2)
        if opt != "?":
            required.add(field)
    return required


def _extract_required_fields_from_openapi(schema: dict, model_name: str) -> set[str] | None:
    """Look up a schema by name in components.schemas and return required fields."""
    schemas = schema.get("components", {}).get("schemas", {})
    target = schemas.get(model_name)
    if not isinstance(target, dict):
        return None
    return set(target.get("required", []))


def _ts_interfaces_in_module(ts_path: Path) -> set[str]:
    content = ts_path.read_text(encoding="utf-8")
    return set(re.findall(r"interface\s+(\w+)", content))


def compare() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--openapi", required=True, help="Path to OpenAPI JSON dump")
    args = parser.parse_args()

    openapi_path = Path(args.openapi)
    if not openapi_path.is_file():
        print(f"FAIL: {openapi_path} not found", file=sys.stderr)
        return 1
    schema = json.loads(openapi_path.read_text(encoding="utf-8"))

    drift: list[str] = []
    modules = ("world", "workspace", "quality")
    for mod in modules:
        ts_file = SHARED_TS_DIR / f"{mod}.ts"
        if not ts_file.is_file():
            print(f"WARN: {ts_file} missing — skipping module {mod}")
            continue
        for iface in _ts_interfaces_in_module(ts_file):
            ts_required = _extract_required_fields_from_ts(ts_file.read_text(encoding="utf-8"), iface)
            oa_required = _extract_required_fields_from_openapi(schema, iface)
            if oa_required is None:
                continue  # not in OpenAPI — OK (DTO not yet exposed via API)
            if ts_required != oa_required:
                drift.append(
                    f"DRIFT {mod}.{iface}: TS required={sorted(ts_required)} "
                    f"!= OpenAPI required={sorted(oa_required)}"
                )

    if drift:
        print("zod reverse validation FAILED:")
        for d in drift:
            print(f"  - {d}")
        return 1
    print("zod reverse validation OK (no drift detected)")
    return 0


def main() -> int:
    return compare()


if __name__ == "__main__":
    sys.exit(main())
```
